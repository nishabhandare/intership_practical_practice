-- Part A - Database & Table Creation
create database banking_db;
use banking_db;
create table accounts(
account_id INT PRIMARY KEY,
account_number VARCHAR(20),
customer_name VARCHAR(100),
account_type VARCHAR(30),
branch VARCHAR(50),
city VARCHAR(50),
balance DECIMAL(12,2),
credit_score INT,
status VARCHAR(20),
opened_date DATE);
describe accounts;
show tables;
